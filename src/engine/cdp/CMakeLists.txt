#-----------------------------------------------------------------------------
# PatchKnob_cdp -- Composer's Desktop Project as a LIBRARY.
#
# CDP ships as 244 command-line programs (400 .c files, 221 of them a main()):
# every transformation is welded to argv parsing and file I/O, and there is no
# callable DSP entry point anywhere in the tree.  Wrapping the executables would
# mean spawning a process per edit, so instead the algorithms are lifted out of
# their main()s into functions that work on buffers, behind cdp_process.h.
#
# This target builds:
#   * CDP's support libraries (soundfile I/O + the PVOC file layer), taken from
#     the vendored tree unchanged, and
#   * the ported transformation sources under processes/.
#
# CDP8 is LGPL 2.1 (vendor/cdp8/LICENSE), which this GPL project may use; the
# vendored tree keeps its own notices and every ported file cites the program it
# came from.
#-----------------------------------------------------------------------------
cmake_minimum_required(VERSION 3.16)
project(PatchKnob_cdp LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CDP_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/cdp8)

if(NOT EXISTS ${CDP_ROOT}/dev/externals/portsf/portsf.c)
    message(WARNING "vendor/cdp8 missing -- PatchKnob_cdp will build empty")
    add_library(PatchKnob_cdp STATIC cdp_process.cpp)
    target_include_directories(PatchKnob_cdp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(PatchKnob_cdp PUBLIC PATCHKNOB_CDP_AVAILABLE=0)
    return()
endif()

# --- CDP support libraries, vendored sources, built as-is -------------------
# NOTE: CDP's own top-level CMakeLists puts -stdlib=libc++ (a Clang flag) into
# CMAKE_CXX_FLAGS_RELEASE, which breaks its C++ programs under GCC/MinGW.  We do
# not use that file, so the flag never reaches us; the C sources below are the
# only vendored code we compile.
set(CDP_SUPPORT_SRCS
    ${CDP_ROOT}/dev/externals/portsf/portsf.c
    ${CDP_ROOT}/dev/externals/portsf/ieee80.c
    ${CDP_ROOT}/dev/pvxio2/pvfileio.c
)

# Ported families are picked up by GLOB and their registrars generated below, so
# adding a port is ONE new file -- no shared registry or CMake list to edit (and
# therefore nothing for parallel work to collide on).  Each processes/cdp_<x>.cpp
# must define  void register_<x>_processes().
file(GLOB CDP_PROCESS_SRCS CONFIGURE_DEPENDS
     ${CMAKE_CURRENT_SOURCE_DIR}/processes/cdp_*.cpp)

set(CDP_REG_DECLS "")
set(CDP_REG_CALLS "")
foreach(src ${CDP_PROCESS_SRCS})
    get_filename_component(fam ${src} NAME_WE)          # cdp_<family>
    string(REGEX REPLACE "^cdp_" "" fam ${fam})
    string(APPEND CDP_REG_DECLS "void register_${fam}_processes();
")
    string(APPEND CDP_REG_CALLS "    register_${fam}_processes();
")
endforeach()
set(CDP_GENERATED_REG ${CMAKE_CURRENT_BINARY_DIR}/cdp_generated_registry.cpp)
file(WRITE ${CDP_GENERATED_REG}
"// GENERATED by src/engine/cdp/CMakeLists.txt -- do not edit.
"
"namespace PatchKnob { namespace cdp {
"
"${CDP_REG_DECLS}"
"void register_ported_processes() {
"
"${CDP_REG_CALLS}"
"}
} }
")

add_library(PatchKnob_cdp STATIC
    cdp_process.cpp
    cdp_graph.cpp
    cdp_pvoc.cpp
    ${CDP_PROCESS_SRCS}
    ${CDP_GENERATED_REG}
    ${CDP_SUPPORT_SRCS}
)

target_include_directories(PatchKnob_cdp PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CDP_ROOT}/dev/externals/include
    ${CDP_ROOT}/dev/newinclude
    ${CDP_ROOT}/include
)

target_compile_definitions(PatchKnob_cdp PUBLIC PATCHKNOB_CDP_AVAILABLE=1)
if(WIN32)
    target_compile_definitions(PatchKnob_cdp PRIVATE WIN32 _WIN32)
endif()
target_compile_options(PatchKnob_cdp PRIVATE -O2)

# --- headless self-test -----------------------------------------------------
add_executable(cdp_test cdp_test.cpp)
target_link_libraries(cdp_test PRIVATE PatchKnob_cdp)
enable_testing()
add_test(NAME cdp_test COMMAND cdp_test)
